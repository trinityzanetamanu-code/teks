                        var CCPlayerState = {
                            player: null,
                            playerId: '',
                            restoreToken: 0,
                            storageKey: 'cc_player_preferences_v1'
                        };

                        function CCPlayerApi(player, command, value) {
                            try {
                                if (!player || typeof player.api !== 'function') return null;
                                if (arguments.length > 2) return player.api(command, value);
                                return player.api(command);
                            } catch (error) {
                                return null;
                            }
                        }

                        function CCPlayerGet() {
                            if (CCPlayerState.player && typeof CCPlayerState.player.api === 'function') {
                                return CCPlayerState.player;
                            }

                            if (CCPlayerState.playerId) {
                                var player = window[CCPlayerState.playerId.replace('-', '_')];
                                if (player && typeof player.api === 'function') {
                                    CCPlayerState.player = player;
                                    return player;
                                }
                            }

                            return null;
                        }

                        function CCPlayerNormalize(value) {
                            return String(value == null ? '' : value)
                                .toLowerCase()
                                .replace(/\s+/g, ' ')
                                .trim();
                        }

                        function CCPlayerOptionTitle(option) {
                            if (typeof option === 'string' || typeof option === 'number') {
                                return String(option);
                            }

                            if (!option || typeof option !== 'object') return '';

                            return String(
                                option.title != null ? option.title :
                                option.name != null ? option.name :
                                option.label != null ? option.label :
                                option.value != null ? option.value : ''
                            );
                        }

                        function CCPlayerOptions(player, command) {
                            var options = CCPlayerApi(player, command);

                            if (Array.isArray(options)) return options;
                            if (options && Array.isArray(options.items)) return options.items;
                            if (options && Array.isArray(options.list)) return options.list;

                            return [];
                        }

                        function CCPlayerFindOption(options, selected, type) {
                            var target = CCPlayerNormalize(selected);
                            if (!target) return -1;

                            var titles = options.map(function(option) {
                                return CCPlayerNormalize(CCPlayerOptionTitle(option));
                            });

                            var index = titles.indexOf(target);
                            if (index !== -1) return index;

                            if (type === 'quality' && target.indexOf('auto') === 0) {
                                index = titles.findIndex(function(title) {
                                    return title === 'auto' || title.indexOf('auto ') === 0;
                                });
                                if (index !== -1) return index;
                            }

                            return titles.findIndex(function(title) {
                                return title && (title.indexOf(target) !== -1 || target.indexOf(title) !== -1);
                            });
                        }

                        function CCPlayerReadPreferences(player) {
                            var subtitle = CCPlayerApi(player, 'subtitle');
                            var muted = CCPlayerApi(player, 'muted');

                            if (subtitle === null || subtitle === false || subtitle === '') {
                                subtitle = -1;
                            }

                            if (muted === 1) muted = true;
                            if (muted === 0) muted = false;

                            return {
                                audiotrack: CCPlayerApi(player, 'audiotrack'),
                                subtitle: subtitle,
                                quality: CCPlayerApi(player, 'quality'),
                                volume: CCPlayerApi(player, 'volume'),
                                muted: muted
                            };
                        }

                        function CCPlayerSavePreferences(player) {
                            var preferences = CCPlayerReadPreferences(player);

                            try {
                                localStorage.setItem(CCPlayerState.storageKey, JSON.stringify(preferences));
                            } catch (error) {}

                            return preferences;
                        }

                        function CCPlayerStoredPreferences() {
                            try {
                                var stored = localStorage.getItem(CCPlayerState.storageKey);
                                return stored ? JSON.parse(stored) : null;
                            } catch (error) {
                                return null;
                            }
                        }

                        var normalized = CCPlayerNormalize(value);

                        return normalized === '' ||
                        normalized === 'off' ||
                        normalized === 'none' ||
                        normalized === 'disabled';
                        }
						
					
                        function CCPlayerRestorePreferences(player, preferences) {
                            if (!player || !preferences) return;

                            var token = ++CCPlayerState.restoreToken;
                            var attempts = 0;
                            var restored = {
                                volume: !Number.isFinite(Number(preferences.volume)),
                                muted: typeof preferences.muted !== 'boolean',
                                audiotrack: preferences.audiotrack == null,
                                subtitle: preferences.subtitle == null,
                                quality: preferences.quality == null
                            };

                            function restore() {
                                if (token !== CCPlayerState.restoreToken) return;
                                attempts++;

                            /*
                            * After a large seek on iOS, PlayerJS/AVPlayer may
                            * recreate the internal player instance.
                            * On each attempt, retrieve the current object.
                            */
                                var currentPlayer = CCPlayerGet();
                                if (currentPlayer && typeof currentPlayer.api === 'function') {
                                    player = currentPlayer;
                                }

                                if (!restored.volume && Number.isFinite(Number(preferences.volume))) {
                                    CCPlayerApi(player, 'volume', Math.max(0, Math.min(1, Number(preferences.volume))));
                                    restored.volume = true;
                                }

                                if (!restored.muted && typeof preferences.muted === 'boolean') {
                                    CCPlayerApi(player, preferences.muted ? 'mute' : 'unmute');
                                    restored.muted = true;
                                }

                                if (!restored.audiotrack && preferences.audiotrack != null) {
                                    var audioOptions = CCPlayerOptions(player, 'audiotracks');
                                    var audioIndex = CCPlayerFindOption(audioOptions, preferences.audiotrack, 'audiotrack');
                                    if (audioIndex !== -1) {
                                        CCPlayerApi(player, 'audiotrack', audioIndex);
                                        restored.audiotrack = true;
                                    }
                                }

                                if (!restored.quality && preferences.quality != null) {
                                    var qualityOptions = CCPlayerOptions(player, 'qualities');
                                    var qualityIndex = CCPlayerFindOption(qualityOptions, preferences.quality, 'quality');
                                    if (qualityIndex !== -1) {
                                        CCPlayerApi(player, 'quality', qualityIndex);
                                        restored.quality = true;
                                    }
                                }

                                if (!restored.subtitle && preferences.subtitle != null) {
                                    if (CCPlayerSubtitleDisabled(preferences.subtitle)) {
                                        CCPlayerApi(player, 'subtitle', -1);
                                        restored.subtitle = true;
                                    } else {
                                        var subtitleOptions = CCPlayerOptions(player, 'subtitles');
                                        var subtitleIndex = CCPlayerFindOption(subtitleOptions, preferences.subtitle, 'subtitle');
                                        if (subtitleIndex !== -1) {
                                            CCPlayerApi(player, 'subtitle', subtitleIndex);
                                            restored.subtitle = true;
                                        }
                                    }
                                }

                                var complete = restored.volume && restored.muted && restored.audiotrack &&
                                    restored.subtitle && restored.quality;

                                if (!complete && attempts < 30) {
                                    setTimeout(restore, 300);
                                }
                            }

                            setTimeout(restore, 300);
                        }

                        function PlayerjsReady(id) {
                            var player = window[id.replace('-', '_')];

                            CCPlayerState.playerId = id;
                            CCPlayerState.player = player;

                            if (typeof window.CCContinueWatchingAttach === 'function') {
                                window.CCContinueWatchingAttach(player);
                            } else {
                                window.CCContinueWatchingPendingPlayer = player;
                            }

                            var storedPreferences = CCPlayerStoredPreferences();
                            if (storedPreferences) {
                                CCPlayerRestorePreferences(player, storedPreferences);
                            }

                            if (location.hash === '#watch') {
                                player.api('mute');
                                player.api('play');
                                player.api('unmute');
                            }
                        }

	modal.dataset.ccBookmarkBound = '1';
	var CONFIRMED_KEY = 'cc_bookmark_rip_confirmed_until';
	var SNOOZE_KEY = 'cc_bookmark_rip_snooze_until';
	var DAY = 24 * 60 * 60 * 1000;
	var CONFIRMED_PERIOD = 180 * DAY;
	var SNOOZE_PERIOD = 1 * DAY;
	function readTime(key) {
		try {
			return Number(localStorage.getItem(key) || 0);
		} catch (error) {
			return 0;
		}
	}
	function writeTime(key, value) {
		try {
			localStorage.setItem(key, String(value));
		} catch (error) {}
	}
	function openModal() {
		modal.classList.add('is-open');
		modal.setAttribute('aria-hidden', 'false');
		document.body.classList.add('cc-bookmark-modal-open');

		var firstAction = modal.querySelector('.cc-bookmark-modal__open');

		if (firstAction) {
			firstAction.focus();
		}
	}
	function closeModal() {
		modal.classList.remove('is-open');
		modal.setAttribute('aria-hidden', 'true');
		document.body.classList.remove('cc-bookmark-modal-open');
	}
	function remindTomorrow() {
		writeTime(SNOOZE_KEY, Date.now() + SNOOZE_PERIOD);
		closeModal();
	}
	function confirmSaved() {
		writeTime(CONFIRMED_KEY, Date.now() + CONFIRMED_PERIOD);

		try {
			localStorage.removeItem(SNOOZE_KEY);
		} catch (error) {}

		closeModal();
	}
	modal.querySelectorAll('[data-cc-bookmark-later]').forEach(function (element) {
		element.addEventListener('click', function (event) {
			event.preventDefault();
			remindTomorrow();
		});
	});
	var confirmButton = modal.querySelector('[data-cc-bookmark-confirm]');
	if (confirmButton) {
		confirmButton.addEventListener('click', confirmSaved);
	}
	document.addEventListener('keydown', function (event) {
		if (
			event.key === 'Escape'
			&& modal.classList.contains('is-open')
		) {
			remindTomorrow();
		}
	});
	var now = Date.now();
	if (
		readTime(CONFIRMED_KEY) > now
		|| readTime(SNOOZE_KEY) > now
	) {
		return;
	}
	window.setTimeout(openModal, 1800);
